10 REM guessing game
20 CLS
30 N=INT(RND(1)*100)+1
40 T=0
50 PRINT "I AM THINKING OF A NUMBER FROM 1 TO 100"
60 PRINT "GUESS";: INPUT G
70 T=T+1
80 IF G<N THEN PRINT "HIGHER": GOTO 60
90 IF G>N THEN PRINT "LOWER": GOTO 60
100 PRINT "GOT IT IN"; T; "TRIES"
110 FOR I=1 TO 8
120 SOUND 1,200+I*20,5: PAUSE 10
130 NEXT I
140 END
